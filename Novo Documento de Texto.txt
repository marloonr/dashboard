# ─── Data generation ─────────────────────────────────────────────────────────
@st.cache_data
def generate_data():
    random.seed(42)
 
    meses = ["Jan", "Fev", "Mar", "Abr", "Mai", "Jun",
             "Jul", "Ago", "Set", "Out", "Nov", "Dez"]
 
    # Realizado x Orçado (mensal)
    orcado  = [280, 310, 340, 370, 390, 420, 450, 480, 460, 440, 410, 380]
    realizado = [265, 298, 355, 362, 410, 405, 438, 491, 0, 0, 0, 0]  # últimos meses sem dado
 
    df_financeiro = pd.DataFrame({
        "Mês": meses,
        "Orçado (R$k)": orcado,
        "Realizado (R$k)": realizado,
    })
 
    # Evolução física (% conclusão por etapa)
    etapas = [
        "Fundação", "Estrutura", "Alvenaria",
        "Cobertura", "Instalações", "Acabamento", "Paisagismo"
    ]
    previsto   = [100, 100, 90, 75, 60, 30, 10]
    executado  = [100, 98,  82, 70, 48, 15, 0]
 
    df_fisica = pd.DataFrame({
        "Etapa": etapas,
        "Previsto (%)": previsto,
        "Executado (%)": executado,
    })
 
    # Custo por categoria
    categorias = ["Mão de Obra", "Material", "Equipamentos", "Serviços Terceiros", "Encargos"]
    orcado_cat   = [1_200_000, 2_100_000, 450_000, 380_000, 320_000]
    realizado_cat = [1_150_000, 2_280_000, 410_000, 395_000, 305_000]
 
    df_categoria = pd.DataFrame({
        "Categoria": categorias,
        "Orçado": orcado_cat,
        "Realizado": realizado_cat,
    })
    df_categoria["Variação"] = df_categoria["Realizado"] - df_categoria["Orçado"]
    df_categoria["Variação %"] = (df_categoria["Variação"] / df_categoria["Orçado"] * 100).round(1)
 
    # Cronograma de atividades
    today = date.today()
    activities = [
        ("Fundação",      today - timedelta(180), today - timedelta(120), 1.00),
        ("Estrutura",     today - timedelta(130), today - timedelta(50),  0.98),
        ("Alvenaria",     today - timedelta(80),  today + timedelta(20),  0.82),
        ("Cobertura",     today - timedelta(40),  today + timedelta(60),  0.70),
        ("Instalações",   today - timedelta(10),  today + timedelta(100), 0.48),
        ("Acabamento",    today + timedelta(60),  today + timedelta(200), 0.15),
        ("Paisagismo",    today + timedelta(190), today + timedelta(250), 0.00),
    ]
    df_gantt = pd.DataFrame(activities, columns=["Tarefa", "Início", "Fim", "Progresso"])
 
    return df_financeiro, df_fisica, df_categoria, df_gantt
 
 
df_financeiro, df_fisica, df_categoria, df_gantt = generate_data()
